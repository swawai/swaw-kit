# No configuration is normally required. To customize the public-key path, public-key comment, or key-generation algorithm, edit {{ENTRY_FILE}} first.

# Basic usage:
  {{COMMAND}} .help en                Show English help.
  {{COMMAND}} .help zh                Show Chinese help.
  {{COMMAND}} .status key             Show public/private key files, fingerprints, and match state
  {{COMMAND}} .status private         Show the private key and its application state (including ssh-agent)
  {{COMMAND}} .status public          Show the public key and its application state
  {{COMMAND}} .status ssh             Show SSH Client, Server, and default-shell state
  {{COMMAND}} .status                 Show complete state (approximately 2.5 KB of output)
  {{COMMAND}} .status --uac           INTERACTIVE! Same as above, but request UAC elevation

# Key pair:
  {{COMMAND}} .key status              Show public/private key files, fingerprints, and match state
  {{COMMAND}} .key dir                 Open the key-pair directory in File Explorer
  {{COMMAND}} .key gen                 INTERACTIVE! Generate the declared key pair; fail if either file already exists
  {{COMMAND}} .key gen -N              Same as above, but do not prompt; generate a key pair without passphrase protection
  {{COMMAND}} .key fp                  Show the public key's SHA256 fingerprint
  {{COMMAND}} .key delete --yes        Delete both key files; unload an applied private key, and refuse while the public key is applied
  {{COMMAND}} .key delete --yes --uac  INTERACTIVE! Same as above, but request UAC elevation

# Apply the private key for logging in to other machines:
  {{COMMAND}} .private status          Show the private key and its application state (including ssh-agent)
  {{COMMAND}} .private load            INTERACTIVE! May prompt for the private-key passphrase; load the private key into the current user's ssh-agent
  {{COMMAND}} .private load --uac      INTERACTIVE! Same as above, but request UAC elevation
  {{COMMAND}} .private unload          Revoke the private key (remove it from ssh-agent's key list)
  {{COMMAND}} .private unload --uac    INTERACTIVE! Same as above, but request UAC elevation

# Apply the public key for logging in to this machine from elsewhere:
  {{COMMAND}} .public grant            Apply the public key; non-administrator account: %USERPROFILE%\.ssh\authorized_keys, authorizes only the current account
                                          otherwise: C:\ProgramData\ssh\administrators_authorized_keys, authorizes all administrator accounts
  {{COMMAND}} .public grant --uac      INTERACTIVE! Same as above, but request UAC elevation
  {{COMMAND}} .public revoke           Revoke the public key (the inverse of grant)
  {{COMMAND}} .public revoke --uac     INTERACTIVE! Same as above, but request UAC elevation


# Global management commands (affect other Windows users):
  Client:
  {{COMMAND}} .global client status                  Show OpenSSH Client availability; restricted state is reported as unknown
  {{COMMAND}} .global client install --uac           INTERACTIVE! Install OpenSSH Client

  Server:
  {{COMMAND}} .global server status                  Show sshd service, port, listener, firewall, and default-shell state
  {{COMMAND}} .global server install --uac           INTERACTIVE! Install and initialize sshd (Windows OpenSSH Server)
  {{COMMAND}} .global server uninstall --yes --uac   INTERACTIVE! Uninstall sshd and remove this tool's firewall rules
  {{COMMAND}} .global server start --uac             INTERACTIVE! Start sshd without changing its startup mode or firewall rules
  {{COMMAND}} .global server stop --uac              INTERACTIVE! Stop sshd without changing its startup mode or firewall rules
  {{COMMAND}} .global server shell cmd --uac         INTERACTIVE! Set CMD as the default shell
  {{COMMAND}} .global server shell powershell --uac  INTERACTIVE! Set PowerShell as the default shell

  Port and firewall:
  {{COMMAND}} .global server port set <port> --uac   INTERACTIVE! Set the single listening port, reconcile firewall, restart, and verify sshd
  {{COMMAND}} .global server firewall status         Show firewall allowance for sshd's current port
  {{COMMAND}} .global server firewall allow --uac    INTERACTIVE! Allow sshd's current port
  {{COMMAND}} .global server firewall remove --uac   INTERACTIVE! Remove sshd firewall rules owned by this tool
