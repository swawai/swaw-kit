# Git identity wrapper:
  {{COMMAND}}               Open the default terminal with the current Git identity.
  {{COMMAND}} .help zh      Show Chinese help.
  {{COMMAND}} .help en      Show English help.
  {{COMMAND}} .info         Show the current configuration and effective Git identity.


# Sync to the current repository:
  {{COMMAND}} .sync              Persist the current identity to the current repository's .git/config.
  {{COMMAND}} .sync --dry-run    Show what would be written without changing .git/config.
  {{COMMAND}} .sync --clear      Clear values written by the last .sync if they have not been changed.

# Editor and terminal launchers:
  {{COMMAND}}               Open the configured default terminal with the current Git identity.
  {{COMMAND}} .code .       Open the current directory in VS Code with the current Git identity.
  {{COMMAND}} .cursor .     Start Cursor with the current Git identity.
  {{COMMAND}} .cmd          Open cmd.exe with the current Git identity.
  {{COMMAND}} .ps           Open Windows PowerShell 5 with the current Git identity (legacy version included with Windows).
  {{COMMAND}} .pwsh         Open PowerShell 7+ with the current Git identity (installation required).

# Custom commands start with a dot. Other commands are passed through to git after identity injection, for example:
  {{COMMAND}} status
  {{COMMAND}} commit -m "message"
  {{COMMAND}} push
