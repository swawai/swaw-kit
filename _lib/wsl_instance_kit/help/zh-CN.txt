# 基本用法:
  {{COMMAND}}                       进入此命令绑定的 WSL 实例             ({{ENTRY_FILE}} 中的 {WSL_name} 指定)
  {{COMMAND}} .status               显示此实例状态 (例如是否已安装)
  {{COMMAND}} .doctor               检查 WSL、入口配置、实例、存储和网速，整体耗时可能接近 30 秒
  {{COMMAND}} .status --json        同 .status，JSON 格式输出，更利于 Agent 处理
  {{COMMAND}} .status --json --brief 面向 Agent 首轮判断的紧凑 JSON
  {{COMMAND}} .doctor --json        同 .doctor，输出约 6 KB
  {{COMMAND}} .help en              Show English help information, output: ~4.3 KB
  {{COMMAND}} .help zh              显示中文版帮助信息，输出约 4.3 KB

# Agent / 自动化使用提示:
  --dry-run 可以试运行；--uac 会进入交互界面（慎用）；Linux 命令透传请使用 {{COMMAND}} -- <cmd>

# 安装/还原、重装、备份、迁移、删除:
  {{COMMAND}} .install              安装：使用 {WSL_source} 创建此实例    (wsl --import 或 wsl --install)
  {{COMMAND}} .install D:\x.tar     还原：从归档创建/重建此实例           (wsl --import)
  {{COMMAND}} .install --yes        重装：使用 {WSL_source} 重建此实例    (wsl --import 或 wsl --install)
  {{COMMAND}} .install --dry-run    预览安装命令，不实际执行
  {{COMMAND}} .backup               按 {WSL_export_format} 在备份目录导出 (wsl --export)
  {{COMMAND}} .backup D:\x.tar      按 {WSL_export_format} 导出到指定路径 (wsl --export)
  {{COMMAND}} .backup list          列出此实例备份目录中的归档
  {{COMMAND}} .relocate             迁移安装位置到 {WSL_install_dir}     （先在 {{ENTRY_FILE}} 中修改此值）
  {{COMMAND}} .relocate --dry-run   预览迁移命令，不实际执行
  {{COMMAND}} .delete --yes         注销并删除此实例                      (wsl --unregister)
  {{COMMAND}} .delete --yes --uac   清理端口策略需管理员权限时，弹出 UAC
# 停止与保活:
  {{COMMAND}} .t                    停止此实例                            (wsl --terminate)
  {{COMMAND}} .alive 600            让此实例保持后台运行 600 秒           (wsl sleep)
  {{COMMAND}} .alive                保持后台运行，并在当前 Windows 账户下次登录时继续保活
  {{COMMAND}} .alive off            关闭 alive 设置
  {{COMMAND}} .alive status         显示保活相关设置状态
# systemd、sshd 与用户:
  {{COMMAND}} .systemd status       显示此实例 systemd 状态
  {{COMMAND}} .systemd enable       启用此实例 systemd                    (/etc/wsl.conf)
  {{COMMAND}} .systemd disable      禁用此实例 systemd                    (/etc/wsl.conf)
  {{COMMAND}} .sshd status          显示此实例 SSH 服务状态
  {{COMMAND}} .sshd enable 2222     开启 SSH 服务并设置端口               (基于 systemd 托管)
  {{COMMAND}} .sshd disable         停止 SSH 并移除配置公钥
  {{COMMAND}} .user ensure          创建/确认 {WSL_user} 用户已存在       (wsl .. -- sh -lc useradd/adduser)
  {{COMMAND}} .user passwd          交互式设置 {WSL_user} 的密码          (wsl .. -- passwd)
  {{COMMAND}} .user passwd --env VAR 从环境变量设置 {WSL_user} 密码       (wsl .. -- chpasswd)
  {{COMMAND}} .user default         将 {WSL_user} 设置为默认用户          (wsl --manage set-default-user)
# 服务端口管理:
  {{COMMAND}} .port status          显示端口映射、防火墙和网络模式状态
  {{COMMAND}} .port expose 8080     映射 TCP 8080:8080 并让防火墙放行     (需要管理员权限)
  {{COMMAND}} .port expose 8080 --uac 映射端口并弹出 UAC 提权确认
  {{COMMAND}} .port expose 8080 80  映射 8080 到 WSL:80 并放行 (NAT)
  {{COMMAND}} .port del 8080        移除 TCP 8080 的映射与防火墙规则
  {{COMMAND}} .port sync            WSL 的 IP 可能会变 (NAT)，此命令可以刷新
# 快速打开目标目录:
  {{COMMAND}} .dir install          打开此实例 Windows 安装目录
  {{COMMAND}} .dir backup           打开此实例备份目录
  {{COMMAND}} .dir downloads        打开 fallback 镜像缓存目录
  {{COMMAND}} .dir config           打开实例级配置 wsl.conf 所在目录      (explorer \\wsl...)
  {{COMMAND}} .dir ssh              打开此实例中的 /etc/ssh 配置目录
# IDE 编辑:
  {{COMMAND}} .code                 用 VS Code 打开 {WSL_default_workdir} (code --remote=wsl+..)
  {{COMMAND}} .code ~               打开此实例用户主目录
  {{COMMAND}} .cursor /etc          用 Cursor 打开 WSL 绝对路径
  {{COMMAND}} .cursor app/          打开 {WSL_default_workdir}/app/

# WSL 底层虚拟机管理 (会影响当前 Windows 用户下的所有实例):
  {{COMMAND}} .vm                   打开 WSL Settings 可视化配置程序
  {{COMMAND}} .vm status            显示当前用户 WSL VM/全部实例状态
  {{COMMAND}} .vm show              打开 WSL 欢迎/功能介绍页面
  {{COMMAND}} .vm default           将此实例设置为当前用户默认实例        (wsl --set-default)
  {{COMMAND}} .vm -s                关闭当前用户的 WSL 2 VM/全部实例      (wsl --shutdown)
  {{COMMAND}} .vm alive             列出当前用户下所有 WSL alive 任务，并自动清理失效项
  {{COMMAND}} .vm alive del <task>  删除指定 WSL alive 任务
  {{COMMAND}} .vm port              列出当前用户下所有 WSL 端口策略，并在有权限时自动清理失效项
  {{COMMAND}} .vm port --uac        同上，但会交互式提权
  {{COMMAND}} .vm port del <id>     删除指定 WSL 端口策略                 (需要管理员权限)

# 上面未出现的命令，会添加 -d -u --cd 信息后，直接透传给 wsl.exe，例如：
  {{COMMAND}} uname -a              等同 wsl.exe -d WSL_name -u WSL_user --cd WSL_default_workdir uname -a
  {{COMMAND}} -- pwd                等同 wsl.exe -d WSL_name -u WSL_user --cd WSL_default_workdir -- pwd
  {{COMMAND}} --cd /tmp -- pwd      同上，但追加 --cd 参数覆盖默认目录
  {{COMMAND}} -u root -- whoami     同上，但追加 -u 参数覆盖默认用户
