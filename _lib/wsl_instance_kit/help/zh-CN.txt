# 基本用法:
  {{COMMAND}}                       进入此命令绑定的 WSL 实例 ({WSL_name} 定义)
  {{COMMAND}} status                显示此实例状态
  {{COMMAND}} --help zh             显示中文版帮助信息
  {{COMMAND}} --help en             Show English help information
  {{COMMAND}} -h zh                 显示中文版帮助信息
  {{COMMAND}} -h en                 Show English help information
  {{COMMAND}} /? zh                 显示中文版帮助信息
  {{COMMAND}} /? en                 Show English help information


# 管理命令:
  {{COMMAND}} ctl install           使用 {WSL_source} 安装该实例          (wsl --import 或 wsl --install)
  {{COMMAND}} ctl install --dry-run 预览安装命令，不实际执行
  {{COMMAND}} ctl install --fallback 同ctl install 但不依赖 Github 资源   (wsl --install --from-file)
  {{COMMAND}} ctl backup            按 {WSL_export_format} 在备份目录导出 (wsl --export)
  {{COMMAND}} ctl export D:\x.tar   按 {WSL_export_format} 导出到指定路径 (wsl --export)
  {{COMMAND}} ctl config            打开此实例 /etc 目录                  (explorer \\wsl...)
  {{COMMAND}} ctl user ensure       创建/确认 {WSL_user} 用户已存在       (wsl .. -- sh -lc useradd/adduser)
  {{COMMAND}} ctl user default      将 {WSL_user} 设置为默认用户          (wsl --manage set-default-user)
  {{COMMAND}} ctl default           将此实例设置为默认实例                (wsl --set-default)
  {{COMMAND}} ctl terminate         终止此实例                            (wsl --terminate)
  {{COMMAND}} ctl -t                同上，终止此实例                      (wsl --terminate)
  {{COMMAND}} ctl systemd enable    启用此实例 systemd                    (/etc/wsl.conf)
  {{COMMAND}} ctl systemd disable   禁用此实例 systemd                    (/etc/wsl.conf)
  {{COMMAND}} ctl ssh status        显示此实例 SSH 服务状态
  {{COMMAND}} ctl ssh config        打开实例级配置 wsl.conf 所在目录
  {{COMMAND}} ctl ssh enable        开启 SSH 服务 (基于systemd托管)
  {{COMMAND}} ctl ssh disable       停止 SSH 并移除配置公钥               (systemctl/service)
  {{COMMAND}} ctl remove --yes      注销并删除此实例                      (wsl --unregister)

# 全局管理命令(会影响当前 Windows 用户下的所有实例):
  {{COMMAND}} ctl global config     打开用户级配置 .wslconfig 所在目录    (explorer %USERPROFILE%)
  {{COMMAND}} ctl global shutdown   关闭当前用户的 WSL2 VM/全部实例       (wsl --shutdown)
  {{COMMAND}} ctl global -t         同 global shutdown
  {{COMMAND}} ctl global network    按 WSL_network_* 更新 .wslconfig      (%USERPROFILE%)

# 远程编辑:
  {{COMMAND}} code                  用 VSCode 打开{WSL_default_workdir}   (code --remote=wsl+..)
  {{COMMAND}} code ~                打开此实例用户家目录
  {{COMMAND}} cursor /etc           用 Cursor 打开 WSL 绝对路径
  {{COMMAND}} cursor app/           打开 {WSL_default_workdir}/app/


# 上面未出现的命令，会添加 -d -u --cd 信息后，透传给 wsl.exe ，例如：
  {{COMMAND}} uname -a              等同 wsl.exe -d WSL_name -u WSL_user --cd WSL_default_workdir uname -a
  {{COMMAND}} -- pwd                等同 wsl.exe -d WSL_name -u WSL_user --cd WSL_default_workdir -- pwd
  {{COMMAND}} --cd /tmp -- pwd      同上，但追加 --cd 参数覆盖
  {{COMMAND}} -u root -- whoami     同上，但追加 -u 参数覆盖
